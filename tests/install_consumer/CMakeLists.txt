cmake_minimum_required(VERSION 3.16)
project(sjson_consumer LANGUAGES C CXX)
find_package(sJson CONFIG REQUIRED)
enable_testing()
add_executable(consumer_cpp main.cpp)
target_link_libraries(consumer_cpp PRIVATE sJson::sjson_cpp)
add_test(NAME installed_cpp COMMAND consumer_cpp)
if(TARGET sJson::sjson_ffi)
    add_executable(consumer_ffi main.c)
    target_link_libraries(consumer_ffi PRIVATE sJson::sjson_ffi)
    get_target_property(ffi_type sJson::sjson_ffi TYPE)
    if(WIN32 AND ffi_type STREQUAL "SHARED_LIBRARY")
        add_custom_command(TARGET consumer_ffi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:sJson::sjson_ffi> $<TARGET_FILE_DIR:consumer_ffi>)
    endif()
    add_test(NAME installed_ffi COMMAND consumer_ffi)
endif()
